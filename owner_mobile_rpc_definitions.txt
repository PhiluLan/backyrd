CREATE OR REPLACE FUNCTION "public"."upsert_owner_description_v1"("p_spot_id" "uuid", "p_description" "text", "p_keywords" "text"[] DEFAULT NULL::"text"[]) RETURNS "void"
    LANGUAGE "plpgsql" SECURITY DEFINER
    SET "search_path" TO 'public'
    AS $$
declare
  v_owner uuid;
  v_verified boolean;
begin
  if auth.uid() is null then
    raise exception 'not_authenticated';
  end if;

  select owner_user_id, is_verified into v_owner, v_verified
  from public.spot_descriptions
  where spot_id = p_spot_id;

  if v_owner is null or v_verified is distinct from true or v_owner <> auth.uid() then
    raise exception 'not_owner_or_not_verified';
  end if;

  update public.spot_descriptions
  set owner_description = p_description,
      owner_keywords = p_keywords,
      owner_updated_at = now(),
      updated_at = now()
  where spot_id = p_spot_id;
end $$;
